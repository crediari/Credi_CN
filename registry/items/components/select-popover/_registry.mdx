Pass generic items plus `labelKey` and `valueKey`. Use `onSearchChange` for remote search.

```tsx
import { SelectPopover } from "@/components/select-popover";

<SelectPopover
  items={cities}
  value={city}
  onSelect={setCity}
  labelKey="name"
  valueKey="id"
/>
```
