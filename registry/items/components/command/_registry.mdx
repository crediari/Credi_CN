Search and pick commands, records, or shortcuts from a keyboard-first list.

```tsx
import {
  Command,
  CommandEmpty,
  CommandGroup,
  CommandInput,
  CommandItem,
  CommandList,
} from "@/components/ui/command";

<Command>
  <CommandInput placeholder="Buscar..." />
  <CommandList>
    <CommandEmpty>Nada encontrado.</CommandEmpty>
    <CommandGroup heading="Ações">
      <CommandItem>Novo contrato</CommandItem>
    </CommandGroup>
  </CommandList>
</Command>
```
