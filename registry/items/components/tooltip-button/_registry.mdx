Pass `tooltipContent` plus any `Button` props. Defaults to `ghost` and `icon`.

```tsx
import { TooltipButton } from "@/components/tooltip-button";

<TooltipButton tooltipContent="Editar">
  <PencilIcon />
</TooltipButton>
```
